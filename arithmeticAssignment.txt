let i:int;
let c:char;
let n:number;
let r:[
        let x:int;
    ];
let arr:[5]int;
function main(): void {

    // ----- int -----
    i = 10;

    i += 5;
    log i, '\n';      // 15

    i -= 3;
    log i, '\n';      // 12

    i *= 2;
    log i, '\n';      // 24

    i /= 4;
    log i, '\n';      // 6


    // ----- char -> int promotion -----
    i = 1;

    i += 'A';
    log i, '\n';      // 66

    i *= 'B';
    log i, '\n';      // 4356


    // ----- number -----
    n = 2.5;

    n += 1;
    log n, '\n';      // 3.5

    n *= 2;
    log n, '\n';      // 7.0

    n /= 2;
    log n, '\n';      // 3.5

    n -= 'A';
    log n, '\n';      // -61.5


    // ----- arrays and records -----

    arr[0] = 10;

    arr[0] += 5;
    log arr[0], '\n'; // 15



    r.x = 7;
    r.x *= 3;

    log r.x, '\n';    // 21


    // ----- expressions evaluated once -----
    i = 0;

    arr[i] = 1;
    arr[i] += 5;

    log arr[0], '\n'; // 6


    // ----- semantic errors -----

    // c *= 2.5;      // ERROR
    // i += 2.5;      // ERROR
    // c += 1000;     // ERROR if char=int narrowing forbidden
}
